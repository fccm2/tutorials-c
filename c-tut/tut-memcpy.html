Using the memcpy function

origin exp11.c:

$ gcc -o exp11.exe exp11.c -lc
$ ./exp11.exe
origin:[abcdefghijk]
recopy:[_/-/.]
result:[_/-/.fghijk]

#include <string.h>
#include <stdio.h>

int main()
{
  char s[] = "abcdefghijk";
  char r[] = "_/-/.";
  size_t n = 5;
  printf("origin:[%s]\n", s);
  memcpy(s/*dst*/, r, n);
  printf("recopy:[%s]\n", r);
  printf("result:[%s]\n", s);
  return 0;
}
