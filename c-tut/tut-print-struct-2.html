Printing a Structure

file exp13.c:

#include <stdio.h>

struct _stc {
  int d1;
  float f2;
  char *s;
};

void
print_stc(struct _stc *stc)
{
  printf("st-stc: {\n");
  printf(" %d\n", stc->d1);
  printf(" %g\n", stc->f2);
  printf(" %s\n", stc->s);
  printf("}\n");
}

int
main()
{
  struct _stc stc = { 12, 3.6, "hello," };

  print_stc(&stc);

  return 0;
}

we compile test and run:

$ gcc -o exp13.exe exp13.c
$ ./exp13.exe
st-stc: {
 12
 3.6
 hello,
}

here we print without a pointer to the structure, so the fields are access'd with a dot ..
